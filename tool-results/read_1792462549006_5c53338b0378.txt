     1→/**
     2→ * Provider catalog — declares every provider the gateway knows about,
     3→ * plus factory functions that build an executor instance on demand.
     4→ *
     5→ * Each provider definition is declarative so the dashboard and the router
     6→ * share the same source of truth: base URL, default model list, auth mode,
     7→ * whether it requires an API key, etc.
     8→ */

    22→// ── Types ────────────────────────────────────────────────────────────────────
    23→
    24→export type ProviderCategory =
    25→  | "api_key"
    26→  | "free_tier"
    27→  | "oauth"
    28→  | "self_hosted";
    29→
    30→export type ProviderProtocol = "openai" | "scrape";
    31→
    32→export interface ProviderStaticModels {
    33→  id: string;
    34→  object: "model";
    35→  owned_by: string;
    36→  capabilities?: ModelObject["capabilities"];
    37→}
    38→
    39→export interface ProviderDefinition {
    40→  id: string;
    41→  name: string;
    42→  category: ProviderCategory;
    43→  protocol: ProviderProtocol;
    44→  baseUrl: string;
    45→  /** Sign-up URL shown in the dashboard. */
    46→  webUrl?: string;
    47→  /** Whether an API key / token is required at runtime. */
    48→  requiresApiKey: boolean;
    49→  /** Whether the dashboard allows overriding baseUrl. */
    50→  supportsCustomUrl: boolean;
    51→  /** Status message shown when no credential is configured. */
    52→  statusMessage?: string;
    53→  /** Static model list (used when /models isn't public). */
    54→  models?: ProviderStaticModels[];
    55→  /** Optional alias prefix (e.g. `ds` for DeepSeek). */
    56→  alias?: string;
    57→  /** Custom header builder for providers with non-Bearer auth. */
    58→  buildHeaders?: (base: Record<string, string>) => Record<string, string>;
    59→}

   897→/**
   898→ * Convert a user-defined CustomProviderDefinition (stored in config.json) into
   899→ * a ProviderDefinition that the rest of the gateway can treat as built-in.
   900→ */
   901→function customDefToProviderDef(c: CustomProviderDefinition): ProviderDefinition {
   902→  const requiresApiKey = c.authScheme !== "none";
   903→  const buildHeaders = (base: Record<string, string>): Record<string, string> => {
   904→    if (c.authScheme === "none") return base;
   905→    // The actual key is injected at runtime by the executor from the stored
   906→    // credential; here we only set up the header skeleton so the
   907→    // OpenAIExecutor knows which header to populate.
   908→    return base;
   909→  };
   910→  return {
   911→    id: c.id,
   912→    name: c.name,
   913→    category: "api_key",
   914→    protocol: "openai",
   915→    baseUrl: c.baseUrl,
   916→    requiresApiKey,
   917→    supportsCustomUrl: false,
   918→    statusMessage: requiresApiKey ? `${c.name} API key missing` : undefined,
   919→    models: (c.models ?? []).map((m) => ({
   920→      id: m.id,
   921→      object: "model" as const,
   922→      owned_by: c.id,
   923→      capabilities: m.capabilities as ProviderStaticModels["capabilities"],
   924→    })),
   925→    buildHeaders,
   926→    // Custom providers carry auth metadata in a side channel — the executor
   927→    // builder reads it back from the CustomProviderDefinition at construction.
   928→    // (We piggy-back on the alias field to mark custom providers.)
   929→    alias: undefined,
   930→  };
   931→}
   932→
   933→/** Return provider definitions for every user-defined custom provider. */
   934→export function listCustomProviderDefs(): ProviderDefinition[] {
   935→  try {
   936→    return store.listCustomProviders().map(customDefToProviderDef);
   937→  } catch {
   938→    return [];
   939→  }
   940→}
   941→
   942→/** All known provider definitions: built-in catalog + user-defined customs. */
   943→export function allProviderDefs(): ProviderDefinition[] {
   944→  return [...PROVIDER_CATALOG, ...listCustomProviderDefs()];
   945→}

   947→/** Auth metadata for a custom provider (used by the executor builder). */
   948→export function getCustomProviderAuth(id: string): CustomProviderDefinition | undefined {
   949→  return store.findCustomProvider(id);

   954→export interface ProviderInstance {
   955→  definition: ProviderDefinition;
   956→  apiKey: string;
   957→  baseUrl: string;

   960→/** Build an executor instance from a provider definition + runtime credential. */
   961→export function buildExecutor(
   962→  def: ProviderDefinition,
   963→  apiKey: string,
   964→  baseUrlOverride?: string
   965→): unknown {
   966→  const baseUrl = (baseUrlOverride ?? def.baseUrl).replace(/\/$/, "");
   967→  if (def.protocol === "scrape") {
   968→    if (def.id === "deepseek") {
   969→      return new DeepSeekScraperExecutor({
   970→        id: def.id,
   971→        name: def.name,
   972→        baseUrl,
   973→        token: apiKey,
   974→      });
   975→    }
   976→    // Gemini Web Scraper: cookie-based auth with session persistence.
   977→    if (def.id === "gemini-web") {
   978→      const models = def.models?.map((m) => ({
   979→        id: `${def.alias ?? def.id}/${m.id}`,
   980→        object: "model" as const,
   981→        owned_by: def.alias ?? def.id,
   982→        capabilities: m.capabilities,
   983→      }));
   984→      return new GeminiScraperExecutor({
   985→        id: def.id,
   986→        name: def.name,
   987→        baseUrl,
   988→        token: apiKey, // base64-encoded session string for continuity
   989→        models,
   990→      });
   991→    }
   992→    throw new Error(`Unknown scrape provider: ${def.id}`);
   993→  }
   994→  // OpenCode: bespoke executor (Bearer public free tier + Responses-API models).
   995→  if (def.id === "opencode") {
   996→    return new OpenCodeExecutor({
   997→      id: def.id,
   998→      name: def.name,
   999→      baseUrl,
  1000→      apiKey,
  1001→      models: def.models?.map((m) => ({
  1002→        id: `${def.alias ?? def.id}/${m.id}`,
  1003→        object: "model" as const,
  1004→        owned_by: def.alias ?? def.id,
  1005→        capabilities: m.capabilities,
  1006→      })),
  1007→    });
  1008→  }
  1009→  // Anthropic executor
  1010→  if (def.id === "anthropic") {
  1011→    const models = def.models?.map((m) => ({
  1012→      id: `${def.alias ?? def.id}/${m.id}`,
  1013→      object: "model" as const,
  1014→      owned_by: def.alias ?? def.id,
  1015→      capabilities: m.capabilities,
  1016→    }));
  1017→    return new AnthropicExecutor({
  1018→      id: def.id,
  1019→      name: def.name,
  1020→      baseUrl,
  1021→      apiKey,
  1022→      models,
  1023→    });
  1024→  }
  1025→  // Groq executor
  1026→  if (def.id === "groq") {
  1027→    const models = def.models?.map((m) => ({
  1028→      id: `${def.alias ?? def.id}/${m.id}`,
  1029→      object: "model" as const,
  1030→      owned_by: def.alias ?? def.id,
  1031→      capabilities: m.capabilities,
  1032→    }));
  1033→    return new GroqExecutor({
  1034→      id: def.id,
  1035→      name: def.name,
  1036→      baseUrl,
  1037→      apiKey,
  1038→      models,
  1039→    });
  1040→  }
  1041→  // DeepSeek executor
  1042→  if (def.id === "deepseek") {
  1043→    return new DeepSeekExecutor({
  1044→      id: def.id,
  1045→      name: def.name,
  1046→      baseUrl,
  1047→      apiKey,
  1048→    });
  1049→  }
  1050→  // Fireworks executor
  1051→  if (def.id === "fireworks") {
  1052→    const models = def.models?.map((m) => ({
  1053→      id: `${def.alias ?? def.id}/${m.id}`,
  1054→      object: "model" as const,
  1055→      owned_by: def.alias ?? def.id,
  1056→      capabilities: m.capabilities,
  1057→    }));
  1058→    return new FireworksExecutor({
  1059→      id: def.id,
  1060→      name: def.name,
  1061→      baseUrl,
  1062→      apiKey,
  1063→      models,
  1064→    });
  1065→  }
  1066→  // Default: OpenAI-compatible
  1067→  const models = def.models?.map((m) => ({
  1068→    id: `${def.alias ?? def.id}/${m.id}`,
  1069→    object: "model" as const,
  1070→    owned_by: def.alias ?? def.id,
  1071→    capabilities: m.capabilities,
  1072→  }));
  1073→  // Custom providers may declare a non-Bearer auth scheme — pull the metadata
  1074→  // from the config store and pass it down to the OpenAIExecutor.
  1075→  const customAuth = getCustomProviderAuth(def.id);
  1076→  const buildHeaders = customAuth
  1077→    ? (base: Record<string, string>): Record<string, string> => {
  1078→        if (customAuth.authScheme === "none") {
  1079→          delete base["Authorization"];
  1080→          return base;
  1081→        }
  1082→        if (customAuth.authScheme === "x-api-key") {
  1083→          // Move the Bearer token to x-api-key
  1084→          const token = base["Authorization"]?.startsWith("Bearer ")
  1085→            ? base["Authorization"].slice(7)
  1086→            : (customAuth as unknown as { apiKey?: string }).apiKey ?? "";
  1087→          delete base["Authorization"];
  1088→          base["x-api-key"] = token;
  1089→          return base;
  1090→        }
  1091→        if (customAuth.authScheme === "custom-header" && customAuth.authHeaderName) {
  1092→          const token = base["Authorization"]?.startsWith("Bearer ")
  1093→            ? base["Authorization"].slice(7)
  1094→            : (customAuth as unknown as { apiKey?: string }).apiKey ?? "";
  1095→          delete base["Authorization"];
  1096→          const prefix = customAuth.authHeaderPrefix ?? "";
  1097→          base[customAuth.authHeaderName] = prefix ? `${prefix} ${token}`.trim() : token;
  1098→          return base;
  1099→        }
  1100→        return base;
  1101→      }
  1102→    : undefined;
  1103→  return new OpenAIExecutor({
  1104→    id: def.id,
  1105→    name: def.name,
  1106→    baseUrl,
  1107→    apiKey,
  1108→    models,
  1109→    stripProviderPrefix: true,
  1110→    buildHeaders,
  1111→  });
  1112→}

  1114→/** Find a provider definition by id or alias (e.g. "ds" → deepseek). */
  1115→export function findProviderDef(idOrAlias: string): ProviderDefinition | undefined {
  1116→  const lower = idOrAlias.toLowerCase();
  1117→  return allProviderDefs().find(
  1118→    (p) =>
  1119→      p.id.toLowerCase() === lower || p.alias?.toLowerCase() === lower
  1120→  );
  1121→}
